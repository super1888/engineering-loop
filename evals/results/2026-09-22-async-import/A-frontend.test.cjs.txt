const assert = require('node:assert/strict');
const fs = require('node:fs');
const vm = require('node:vm');
const path = require('node:path');
function setup(responses, storage = new Map()) {
  const elements = Object.fromEntries(['rows','status','message','submit','results','retry'].map(id => [id, { value: '', textContent: '', hidden: false, disabled: false, children: [], addEventListener(event, callback) { this[event] = callback; }, replaceChildren() { this.children = []; }, appendChild(child) { this.children.push(child); } }]));
  const calls = [];
  let timer;
  let sequence = 0;
  const context = { localStorage: { getItem: key => storage.get(key), setItem: (key, value) => storage.set(key, value) }, document: { querySelector: s => elements[s.slice(1)], createElement: () => ({ children: [], appendChild(child) { this.children.push(child); } }) }, crypto: { randomUUID: () => `operation-${++sequence}` }, setTimeout: callback => { timer = callback; return 1; }, clearTimeout: () => { timer = null; }, fetch: async (url, options) => { calls.push({url, options}); const data = responses.shift(); if (data instanceof Error) throw data; assert.ok(data, `Unexpected request ${url}`); return { ok: !data.error, status: data.error ? 400 : 200, json: async () => data }; } };
  vm.runInNewContext(fs.readFileSync(path.join(__dirname, '../workspace/ui/app.js'), 'utf8'), context);
  return { elements, calls, storage, tick: () => timer(), settle: () => new Promise(resolve => setImmediate(resolve)) };
}
(async () => {
  const queued = { contractVersion: 2, batchId: 'opaque/batch', status: 'QUEUED', total: 1, succeeded: 0, failed: 0, items: [{rowId: '0001/<b>', status: 'QUEUED', attempts: 0, error: null}] };
  const done = {...queued, status: 'SUCCEEDED', succeeded: 1, items: [{...queued.items[0], status: 'SUCCEEDED', attempts: 1}]};
  const app = setup([queued, done]);
  const input = '[{"rowId":"0001/<b>","name":"Ada"}]';
  app.elements.rows.value = input;
  const sending = app.elements.submit.click();
  assert.equal(app.elements.status.textContent, 'Submitting…');
  await sending;
  assert.match(app.elements.status.textContent, /^QUEUED/);
  assert.equal(app.elements.submit.disabled, true);
  assert.equal(app.elements.results.children[0].children[0].textContent, '0001/<b>');
  await app.tick();
  assert.equal(app.calls[1].url, '/api/batches/opaque%2Fbatch');
  assert.match(app.elements.status.textContent, /^SUCCEEDED/);
  assert.equal(app.elements.submit.disabled, false);
  assert.equal(app.elements.rows.value, input);
  const broken = setup([new Error('connection lost'), queued]);
  broken.elements.rows.value = input;
  await broken.elements.submit.click();
  assert.equal(broken.elements.rows.value, input);
  assert.match(broken.elements.message.textContent, /connection lost/);
  await broken.elements.submit.click();
  assert.equal(JSON.parse(broken.calls[0].options.body).requestId, JSON.parse(broken.calls[1].options.body).requestId);
  const invalid = setup([]);
  invalid.elements.rows.value = '{invalid';
  await invalid.elements.submit.click();
  assert.equal(invalid.elements.rows.value, '{invalid');
  assert.equal(invalid.calls.length, 0);
  const failed = {...queued, status: 'FAILED', failed: 1, items: [{...queued.items[0], status: 'FAILED', attempts: 1, error: 'transient failure'}]};
  const failedAgain = {...failed, items: [{...failed.items[0], attempts: 2}]};
  const failing = setup([failed, new Error('retry receipt lost')]);
  failing.elements.rows.value = input;
  await failing.elements.submit.click();
  assert.equal(failing.elements.retry.hidden, false);
  assert.equal(failing.elements.results.children[0].children[3].textContent, 'transient failure');
  await failing.elements.retry.click();
  assert.equal(failing.elements.rows.value, input);
  const uncertainId = JSON.parse(failing.calls[1].options.body).requestId;
  const reloaded = setup([failedAgain, failedAgain], failing.storage);
  await reloaded.settle();
  assert.match(reloaded.elements.status.textContent, /^FAILED/);
  assert.equal(reloaded.elements.retry.hidden, false);
  assert.equal(JSON.parse(reloaded.storage.get('batch-import-v2')).retryOperation.requestId, uncertainId);
  await reloaded.elements.retry.click();
  assert.equal(JSON.parse(reloaded.calls[1].options.body).requestId, uncertainId);
  assert.equal(JSON.parse(reloaded.storage.get('batch-import-v2')).retryOperation, null);
  assert.equal(reloaded.elements.results.children[0].children[2].textContent, '2');
  assert.equal(reloaded.elements.rows.value, input);
  console.log('PASS: pending/queued visibility, polling, opaque IDs, input preservation, submit identity reuse, failed-item errors/attempts, refresh restoration, uncertain retry ID retained through refresh and terminal GET, replay acknowledgement.');
})().catch(error => { console.error(error); process.exitCode = 1; });
