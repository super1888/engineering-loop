async function start(manual, name) {
  const args = ['server.py', '--db', path.join(temp, name + '.db'), '--port', '0'];
  if (manual) args.push('--manual-worker', '--fail-once-row', 'beta');
  const child = spawn('python', args, {cwd: workspace, stdio: ['pipe', 'pipe', 'pipe'], windowsHide: true});
  servers.push(child);
  const lines = readline.createInterface({input: child.stdout});
  const line = await Promise.race([once(lines, 'line').then(([line]) => line), new Promise((_, reject) => {const timer=setTimeout(() => reject(new Error('server startup timeout')), 10000); timer.unref();})]);
  child.stderr.on('data', chunk => logs.push('server stderr: ' + chunk));
  return {child, url: 'http://127.0.0.1:' + JSON.parse(line).port};
}
async function request(url, method='GET', body) {
  const response = await fetch(url, {method, headers: {'Content-Type':'application/json'}, body: body===undefined ? undefined : JSON.stringify(body)});
  return {status: response.status, body: await response.json()};
}
function checkSnapshot(batch) {
  assert.equal(batch.contractVersion, 2);
  assert.equal(batch.items.length, batch.total);
  assert.equal(batch.succeeded, batch.items.filter(i=>i.status==='SUCCEEDED').length);
  assert.equal(batch.failed, batch.items.filter(i=>i.status==='FAILED').length);
  for (const item of batch.items) {
    assert.equal(typeof item.rowId, 'string');
    assert(Number.isInteger(item.attempts));
    assert(item.error===null || typeof item.error==='string');
  }
}
async function untilSnapshot(url, status) {
  for(let n=0;n<100;n++) {
    const {body} = await request(url);
    checkSnapshot(body);
    if(body.status===status) return body;
    await new Promise(resolve=>setTimeout(resolve,50));
  }
  throw new Error('Timed out awaiting '+status);
}
