async function start(extra=[]) {
  const directory = fs.mkdtempSync(path.join(os.tmpdir(), 'import-a-check-'));
  const db = path.join(directory, 'test.db');
  const child = spawn('python', ['-u', 'server.py', '--db', db, '--port', '0', ...extra], {cwd:workspace, stdio:['pipe','pipe','pipe']});
  child.stderr.on('data', data => log('SERVER STDERR: '+data));
  const port = await new Promise((resolve,reject) => {
    let data='';
    const timer=setTimeout(()=>reject(new Error('server start timeout')),10000);
    child.stdout.on('data', chunk => { data += chunk; if(data.includes('\n')) {clearTimeout(timer); resolve(JSON.parse(data.split('\n')[0]).port);} });
    child.once('error',reject);
  });
  return {child,db,directory,base:`http://127.0.0.1:${port}`};
}
async function stop(server) {
  server.child.kill();
  await new Promise(resolve=>server.child.once('exit',resolve));
  fs.rmSync(server.directory,{recursive:true,force:true});
}
async function api(base, route, body) {
  const response = await fetch(base+route, body === undefined ? {} : {method:'POST',headers:{'content-type':'application/json'},body:JSON.stringify(body)});
  return {status:response.status,body:await response.json()};
}
