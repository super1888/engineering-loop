const assert = require('node:assert/strict');
const { test, before, after } = require('node:test');
const { chromium } = require('playwright');

let browser;
before(async () => {
  browser = await chromium.launch({ executablePath: '<BROWSER>', headless: true });
  console.log(`Browser: ${browser.version()}; Node: ${process.version}`);
});
after(async () => { await browser?.close(); });

async function dismiss(page, action) {
  if (action === 'Escape') await page.keyboard.press('Escape');
  else if (action === 'Backdrop') await page.locator('#detail').click({ position: { x: 5, y: 5 } });
  else await page.getByRole('button', { name: action, exact: true }).click();
}

async function listSnapshot(page) {
  return page.evaluate(() => ({
    group: document.querySelector('#group').value,
    page: document.querySelector('#page-status').textContent,
    scroll: document.querySelector('#records').scrollTop,
    rows: document.querySelectorAll('#records .row').length,
    lists: document.querySelectorAll('#list-view:not([hidden])').length,
    detailHidden: document.querySelector('#detail').hidden,
  }));
}

for (const action of ['Save', 'Cancel', 'Close detail', 'Escape', 'Backdrop']) {
  for (const direct of [false, true]) {
    test(`${direct ? 'Direct detail defaults' : 'List context restored'} via ${action}`, async () => {
      const page = await browser.newPage();
      page.setDefaultTimeout(3000);
      const errors = [];
      page.on('pageerror', error => errors.push(error.message));
      try {
        await page.goto(`http://127.0.0.1:8767/${direct ? '?record=17' : ''}`);
        let expected;
        const id = direct ? 17 : 73;
        if (!direct) {
          await page.locator('#group').selectOption('A');
          await page.getByRole('button', { name: 'Next', exact: true }).click();
          await page.locator('#records').evaluate(element => { element.scrollTop = 300; });
          expected = await listSnapshot(page);
          assert.equal(expected.scroll, 300);
          await page.getByRole('button', { name: `Open record ${id}`, exact: true }).click();
        } else {
          expected = { group: 'all', page: 'Page 1 of 6', scroll: 0, rows: 30, lists: 1, detailHidden: true };
        }
        await page.getByLabel('Name', { exact: true }).fill(' Edited record ');
        await dismiss(page, action);
        assert.deepEqual(await listSnapshot(page), expected);
        const expectedName = action === 'Save' ? 'Edited record' : `Record ${id}`;
        const open = page.getByRole('button', { name: `Open record ${id}`, exact: true });
        assert.equal((await open.locator('..').locator('span').textContent()), expectedName);
        await open.click();
        assert.equal(await page.getByLabel('Name', { exact: true }).inputValue(), expectedName);
        await page.getByLabel('Name', { exact: true }).fill('Discard this second edit');
        await page.getByRole('button', { name: 'Cancel', exact: true }).click();
        assert.equal(await open.locator('..').locator('span').textContent(), expectedName);
        await page.getByRole('button', { name: 'Next', exact: true }).click();
        assert.equal(await page.locator('#page-status').textContent(), direct ? 'Page 2 of 6' : 'Page 3 of 3');
        await page.getByRole('button', { name: 'Previous', exact: true }).click();
        assert.equal(await page.locator('#page-status').textContent(), expected.page);
        assert.deepEqual(errors, []);
      } finally {
        await page.close();
      }
    });
  }
}
